      11                         : // concrete types. The lowering rules are outlined below.
      12                         : //
      13                         : // ByteAddressBuffer/RWByteAddressBuffer/StructuredBuffer/RWStructuredBuffer:
      14                         : //
      15                         : //     struct LoweredType {
      16                         : //         T* data;
      17                         : //         size_t size;
      18                         : //     }
      19                         : //
      20                         : //     Instructions that access `LoweredType` are lowered immediately in this
      21                         : //     pass such that they index `data` as needed.
      22                         : //
      23                         : // ConstantBufferType/ParameterBlockType:
      24                         : //
      25                         : //     typealias LoweredType = T*;
      26                         : //
      27                         : //     Instructions that access `LoweredType` are lowered immediately in this
      28                         : //     pass such that they access members through the pointer as needed.
      29                         : //
      30                         : // All other resource types (Textures, Samplers, RaytracingAccelerationStructures):
      31                         : //
      32                         : //     typealias LoweredType = void*;

      34                         : //     These types are generally accessed only via functions defined in the
      35                         : //     core module using inline assembly. Once support for these operations on
      36                         : //     the LLVM target is added, the inline assembly should expect these types
      37                         : //     to be pointers in the LLVM IR.
      38                         : //
