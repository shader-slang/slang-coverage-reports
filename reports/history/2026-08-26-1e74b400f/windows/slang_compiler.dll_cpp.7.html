 1113    case '\f':
 01114        out << "\\f";
 1115        return;

 01117        out << "\\n";
 1118        return;
 1119    case '\r':
 01120        out << "\\r";

 1125    }
 11126    if (uc < 0x20)
 1127    {
 1128        char buf[8];
 01129        snprintf(buf, sizeof(buf), "\\u%04x", (unsigned)uc);
