 493/// Replays an existing lookup path and applies its substitutions to the selected leaf.
 494///
 495/// A second traversal keeps the structural lookup allocation-free and avoids specializing an
 496/// intermediate witness table, which would copy every entry before selecting the next one.
 497/// This traversal is deliberately paired with `_tryLookUpExistingRequirementWitnessRec`: every
 498/// witness shape accepted by the selection pass must visit the same conformance declarations here,
 499/// in the same inner-to-outer order. The forceful frontier traversal below must recognize that same
 500/// structural vocabulary, although it stops at the first missing step instead of reaching a leaf.
 501/// The shared shapes are a known `PackBranchSubtypeWitness`, a `DeclaredSubtypeWitness` backed by a
 502/// nested lookup, inheritance declaration, or generic constraint, and a `TransitiveSubtypeWitness`
 503/// whose mid-to-super witness is declared. Selection distinguishes the three declared-witness
 504/// sources to find their table; specialization intentionally treats them uniformly because only
 505/// the substitutions on their conformance decl-ref matter at that stage.
 506static RequirementWitness _specializeExistingRequirementWitnessRec(
 507    ASTBuilder* astBuilder,
 508    SubtypeWitness* subtypeWitness,
 509    RequirementWitness requirementWitness)
 1510{

 512    {
 0513        auto selectedWitness = _tryGetKnownPackBranchWitness(packBranchWitness);
 0514        if (!selectedWitness)
 515            return RequirementWitness();
 0516        return _specializeExistingRequirementWitnessRec(
 517            astBuilder,
 518            selectedWitness,
 519            requirementWitness);
 520    }
 521
 1522    if (auto declaredSubtypeWitness = as<DeclaredSubtypeWitness>(subtypeWitness))
 523    {
 1524        auto conformanceDeclRef = declaredSubtypeWitness->getDeclRef();

 526        {
 1527            requirementWitness = _specializeExistingRequirementWitnessRec(
 528                astBuilder,
 529                nestedLookupDeclRef->getWitness(),
 530                requirementWitness);
 531        }

 552/// Passively looks up an existing requirement witness for ordinary value resolution.
 553///
 554/// Forceful semantic clients use `locateNextRequirementWitnessLookupFrontier` instead. Keeping the
 555/// operations separate lets this hot path avoid allocating a specialization list or producing
 556/// missing-frontier state that its callers cannot consume.
 557RequirementWitness tryLookUpRequirementWitness(
 558    ASTBuilder* astBuilder,
 559    SubtypeWitness* subtypeWitness,
 560    Decl* requirementKey)
 1561{
 1562    auto requirementDeclRef = makeDeclRef(requirementKey);
 1563    auto requirementWitness =
 564        _tryLookUpExistingRequirementWitnessRec(astBuilder, subtypeWitness, requirementDeclRef);

 1566    if (requirementWitness.getFlavor() == RequirementWitness::Flavor::none)

 1569            return RequirementWitness(subtypeWitness->getSub());

 1571            return RequirementWitness(subtypeWitness);
 1572        return RequirementWitness();

 583    // the leaf. A fallthrough here means the paired structural traversals have drifted apart.

 741RequirementWitnessLookupFrontier locateNextRequirementWitnessLookupFrontier(
 742    ASTBuilder* astBuilder,
 743    SubtypeWitness* subtypeWitness,
 744    DeclRef<Decl> requirementDeclRef)
 1745{
 1746    if (!subtypeWitness || !requirementDeclRef)
 0747        return RequirementWitnessLookupFrontier();
 748
 1749    auto result = _locateNextRequirementWitnessLookupFrontierRec(
 750        astBuilder,
 751        subtypeWitness,
 752        requirementDeclRef);
 753
 754    // These two entries are structural properties of the original subtype witness. Preserve the
 755    // historical lookup precedence by using a table entry when present and synthesizing the value
 756    // only after traversal fails. This fallback belongs at the public boundary: an intermediate
 757    // missing table on a nested witness path must not hide the original witness's `ThisType`.
 1758    if (result.frontier.getStatus() != RequirementWitnessLookupFrontierStatus::Found)
 759    {
 1760        if (as<ThisTypeDecl>(requirementDeclRef.getDecl()))
 761        {
 0762            result = UnspecializedRequirementWitnessLookupFrontier();
 0763            result.frontier = RequirementWitnessLookupFrontier::makeFound(
 764                RequirementWitness(subtypeWitness->getSub()));
 0765        }
 1766        else if (as<ThisTypeConstraintDecl>(requirementDeclRef.getDecl()))
 767        {
 0768            result = UnspecializedRequirementWitnessLookupFrontier();
 0769            result.frontier =
 770                RequirementWitnessLookupFrontier::makeFound(RequirementWitness(subtypeWitness));
 771        }
 772    }
 773
 0774    if (result.frontier.getStatus() == RequirementWitnessLookupFrontierStatus::Found)
 775    {
 776        // The recursive traversal records each declaration context on the way back out: first the
 777        // context that owns the selected table, then each context whose projected conformance led
 778        // to it. Replay that same inner-to-outer order so every subsequent substitution sees the
 779        // declaration-context result produced by the preceding step.
 1780        auto specializedWitness = _specializeRequirementWitnessAlongLookupPath(
 781            astBuilder,
 782            result.frontier.getWitness(),
 783            result.specializationDeclRefs);
 1784        result.frontier.setFoundWitness(specializedWitness);
 1785    }
 1786    return result.frontier;
