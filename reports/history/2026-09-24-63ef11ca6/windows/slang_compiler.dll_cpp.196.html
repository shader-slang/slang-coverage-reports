 758// Unroll all `[ForceUnroll]` loops in a function, in two passes.
 759//
 760// A `[ForceUnroll]` loop whose trip count depends on an enclosing `[ForceUnroll]` loop's
 761// induction variable cannot be unrolled on its own: while the enclosing loop is still rolled,
 762// that variable is loop-carried, so the inner loop's bound never folds to a constant. Consider
 763// shader-slang/slang#12473:
 764//
 765//     [ForceUnroll] for (int i = 0; i < 8; ++i)
 766//         [ForceUnroll] for (int j = 0; j < i; ++j)
 767//             dst[i*8+j] = src[i*8+j];
 768//
 769// If the `j` loop is attempted first (inner-first, the natural order), `j < i` never folds and
 770// `_unrollLoop` bails without unrolling it. Once the `i` loop is unrolled, each cloned body has
 771// `i` substituted by a per-iteration literal, so the copies of the `j` loop then unroll.
 772//
 773//  - Pass 1 unrolls what it can, inner-first, and does NOT diagnose a loop it fails to unroll —
 774//    it simply leaves it. It runs `_unrollLoop` with `allowCheapBail`, so a loop that has not
 775//    resolved an exit test after one peel is left for pass 2 rather than peeled all 4096 times.
 776//    A premature bail here is harmless: pass 1 emits no diagnostic, so at worst the loop is
 777//    deferred to pass 2, which decides its fate honestly.
 778//  - Pass 2 unrolls outermost-first and DOES diagnose, with `allowCheapBail` off so it peels a
 779//    loop fully (up to the iteration limit) before concluding it cannot terminate. By the time it
 780//    reaches a loop, every enclosing `[ForceUnroll]` loop has already been unrolled (or itself
 781//    diagnosed and aborted), so a loop that still will not unroll here genuinely cannot be, and its
 782//    failure is reported. Because pass 2 aborts on the first failure and each loop is visited once,
 783//    "diagnose exactly once" holds without any per-loop bookkeeping.
 784bool unrollLoopsInFunc(
 785    TargetProgram* targetProgram,
 786    IRModule* module,
 787    IRGlobalValueWithCode* func,
 788    DiagnosticSink* sink,
 789    bool* outChanged)
 1790{
 791    // Nothing to do if this function has no `[ForceUnroll]` loops. Unrolling only ever clones
 792    // existing `[ForceUnroll]` loops (it never introduces the first one), so both passes are
 793    // no-ops here. Returning early also avoids running `sortBlocksInFunc` on a function we did not
 794    // touch, which would needlessly reorder its blocks and perturb otherwise-unrelated emit.
 1795    auto pass1Loops = collectLoopsInFunc(
 796        func,
 1797        [](IRLoop* l) { return l->findDecoration<IRForceUnrollDecoration>() != nullptr; });
 1798    if (pass1Loops.getCount() == 0)
 799        return true;
 800
 801    // Rewrite each loop's `[ForceUnroll]` operand from the user's iteration count into a
 802    // peel-attempt budget so both passes can treat it uniformly as "attempts remaining" and spend
 803    // it down as they peel (see `_normalizeForceUnrollBudgets`). `pass1Loops` already holds exactly
 804    // this function's `[ForceUnroll]` loops, so it is reused rather than re-walking the blocks.
