 918ChunkBuilder* SerialWriter::_getOrCreateChunkForLayout(LayoutObj* layout)
 1919{
 1920    if (!layout)
 1921        return nullptr;
 922
 923    // We start by looking for an existing chunk for `layout`,
 924    // which would be cached on the object itself.
 925    //
 1926    if (auto existingChunk = layout->chunk)
 927        return existingChunk;

 929    // Next we look for an existing chunk that matches the
 930    // structure of `layout`.
 931    //

 934    {
 1935        auto existingChunk = *found;

 938    }
 939

 941    // then we'll create one.
 942    //

 1949    chunk->writeData(&rawKind, sizeof(rawKind));
 950

 952    {
 953    default:
 954        break;
 955
 956    case FossilizedValKind::Ptr:
 957    case FossilizedValKind::OptionalObj:
 958        {
 959            auto containerLayout = (ContainerLayoutObj*)layout;
 960            auto elementLayout = containerLayout->baseLayout;
 1961            auto elementLayoutChunk = _getOrCreateChunkForLayout(elementLayout);
 1962            chunk->writeRelativePtr<FossilInt>(elementLayoutChunk);

 1964        break;
 965
 966    case FossilizedValKind::ArrayObj:
 967    case FossilizedValKind::DictionaryObj:
 968        {
 969            auto containerLayout = (ContainerLayoutObj*)layout;
 1970            auto elementLayout = containerLayout->baseLayout;
 1971            auto elementLayoutChunk = _getOrCreateChunkForLayout(elementLayout);

 976            {
 1977                elementStride =
 978                    UInt32(roundUpToAlignment(elementLayout->size, elementLayout->alignment));
 979                SLANG_ASSERT(elementStride != 0);
 980            }

 982        }
 1983        break;

 985    case FossilizedValKind::Struct:
 986    case FossilizedValKind::Tuple:

 1997                chunk->writeRelativePtr<FossilInt>(fieldLayoutChunk);

 1007                    //

 1013    }

 1172
 1173SerialReader::~SerialReader()
 11174{
 1175    // If an application is designed to perform something
 1176    // like on-demand deserialization, it may create
 1177    // additional `SerialReader`s attached to the same
 1178    // `ReadContext`, potentially even in the body of a
 1179    // callback that was invoked by an operation on another
 1180    // `SerialReader` further up the stack.
 1181    //
 1182    // If we were to track the deferred actions that get
 1183    // enqueued on a per-`SerialReader` basis, and then
 1184    // flush them when the given `SerialReader` is destructed,
 1185    // it could potentially lead to very deep call stacks.
 1186    //
 1187    // Instead, we track a single list of deferred actions
 1188    // on the `ReadContext`, which means that we need to
 1189    // figure out when to actually flush that list.
 1190    //
 1191    // What is implemented here is a "last one out shuts the door"
