 718Index CompletionContext::determineCompletionItemSortOrder(
 719    Decl* item,
 720    const List<Type*>& expectedTypes)
 1721{
 1722    if (expectedTypes.getCount() == 0)
 1723        return -1;
 724
 725    // Test if `itemType` matches `expectedType`, and return the relevance of the match.
 726    // -1 means no match, a positive number means a match.
 727    // The smaller the number, the more relevant the match is, and the item will be listed
 728    // earlier in the completion list.
 729    auto matchType = [&](Type* itemType, DeclRefType* expectedType) -> Index
 1730    {
 1731        if (itemType == expectedType)
 1732            return 1; // Exact match

 1734        auto declRef = isDeclRefTypeOf<Decl>(itemType);
 1735        if (!declRef)
 736            return -1; // No match
 737
 1738        if (declRef.getDecl() == expectedType->getDeclRef().getDecl())
 1739            return 2; // Match by decl
 740
 741        // We may also want to extend the matching logic to include subtyping or other
 742        // coercion relationships. But for now, we will just check for simple matches
 743        // to avoid performance problems.
 744        //
 0745        return -1;
 1746    };
 747
 748    Index result = -1;

 750    // If we have any expected types, we will sort the completion candiate items by their relevance

 761        {
 1762            if (as<EnumDecl>(item))
 1763                currentSortOrder = 0;
 1764            else if (!as<InterfaceDecl>(item))
 1765                currentSortOrder = 1;
 1766        }
 1767        else if (auto varItem = as<VarDeclBase>(item))
 768        {
 1769            currentSortOrder = matchType(varItem->type.type, etDeclRefType);
 1770        }
 1771        else if (auto callableItem = as<CallableDecl>(item))
 772        {
 773            // If the item is a callable decl, we will check if the return type matches the expected
 774            // type.
 1775            currentSortOrder = matchType(callableItem->returnType.type, etDeclRefType);
 776        }
 1777        if (result == -1 || (currentSortOrder != -1 && currentSortOrder < result))
 778        {
 779            // If we have a better match, we will update the result.
 1780            result = currentSortOrder;

 783    // Always list decls within the same module first.
 784    // Note if result == 0, it means the item is representing the expected enum type itself,
 785    // so we always want to list it first by not increasing `result`.
 1786    if (result > 0 && getModule(item) != parsedModule)
 1787        result++;

 1789    if (result > 0 && isFromCoreModule(item))
 1790        result++;
 1791    return result;
