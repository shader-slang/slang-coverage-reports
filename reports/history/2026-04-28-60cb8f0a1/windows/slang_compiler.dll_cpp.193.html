 680LivenessContext::BlockResult LivenessContext::_processBlock(
 681    BlockIndex blockIndex,
 682    const ConstArrayView<IRInst*>& run,
 683    const Loop* loop)
 0684{

 686    //
 687    // * If root start block - before the start (if accessed via successor)
 688    // * If root start block - after the start (if accessed initially in search)
 689    // * Otherwise the whole run for the block
 690    //
 691    // Since this is the case, we know start is not part of the run
 692    SLANG_ASSERT(run.indexOf(m_rootLiveStart) < 0);
 693
 694    // If there is *another* start to the same root, we can't traverse to other blocks, and the last
 695    // access in this block must be the result
 696    {
 697        // NOTE! We shouldn't/can't use run.indexOf here, because we are looking for *any* start to
 698        // the root _indexOfRootStart does this search. Moreover we know (it's a condition on run
 699        // passed into this function) run cannot contain the root start.
 0700        const Index startIndex = _indexOfRootStart(run);
 0701        if (startIndex >= 0)
 702        {
 703            // Complete the block with this run
 0704            return _completeBlock(blockIndex, run.head(startIndex));
 705        }
 706    }
 707
 708    // Find all the successors for this block
 0709    auto successors = _getSuccessors(blockIndex);
 710
 0711    const Index successorCount = successors.getCount();
 712
 713    // NOTE! Care is needed around successorResults, because _processorSuccessor may cause the
 714    // underlying list to be reallocated. If we always access through successorResults (ie
 715    // RAIIStackArray type), things will be fine though.
 716
 717    // Set up space to store successor results
 0718    RAIIStackArray<BlockResult> successorResults(&m_successorResults);
 0719    successorResults.setCount(successorCount);
 720
 721    // If we hit a loop add the information and make this the current loop info
 722    {
 0723        const auto& fixedInfo = _getFixedBlockInfo(blockIndex);
 0724        if (fixedInfo.isLoopStart())
 725        {
 726            SLANG_ASSERT(_getLoopTerminator(fixedInfo.block));
 727
 728            Loop nextLoop;
 0729            nextLoop.parentLoop = loop;
 0730            nextLoop.breakBlockIndex = fixedInfo.breakBlockIndex;
 0731            nextLoop.targetBlockIndex = fixedInfo.targetBlockIndex;
 0732            nextLoop.loopBlockIndex = blockIndex;
 733
 0734            for (Index i = 0; i < successorCount; ++i)
 735            {
 0736                const auto result = _processSuccessor(successors[i], &nextLoop);
 0737                successorResults[i] = result;
 738            }
 0739        }
 740        else
 741        {
 0742            for (Index i = 0; i < successorCount; ++i)
 743            {
 0744                const auto result = _processSuccessor(successors[i], loop);
 0745                successorResults[i] = result;
 746            }
 747        }
 748    }
 749
 750    // Zero initialize all the counts
 0751    Index foundCounts[Index(BlockResult::CountOf)] = {0};
 0752    for (const auto successorResult : successorResults.getConstView())
 753    {
 754        // Change counts depending on the result
 0755        foundCounts[Index(successorResult)]++;
 756    }
 757
 758    const Index foundCount = foundCounts[Index(BlockResult::Found)];
 759    const Index notFoundCount = foundCounts[Index(BlockResult::NotFound)];
 760
 761    const Index otherCount = successorCount - (foundCount + notFoundCount);
 762
 763    // If one or more of the successors (or successors of successors),
 764    // was found to have the last access, we need to mark the end of scope
 765    // at the start of any other paths (which are dominated).
 0766    if (foundCount > 0)
 767    {
 768        // If all successors have result, or are not dominated
 0769        if (foundCount + otherCount == successorCount)
 770        {
 771            return _addBlockResult(blockIndex, BlockResult::Found);
 772        }
 773
 0774        auto successorResultsView = successorResults.getConstView();
 775
 0776        for (Index i = 0; i < successorCount; ++i)
 777        {
 0778            const auto successorResult = successorResultsView[i];
 779
 0780            if (successorResult == BlockResult::NotFound)
 781            {
 0782                const auto successorBlockIndex = successors[i];
 0783                _maybeAddEndAtBlockStart(successorBlockIndex);
 0784                _addBlockResult(successorBlockIndex, BlockResult::Found);
 785            }

 789        return _addBlockResult(blockIndex, BlockResult::Found);
 790    }
 791
 0792    return _completeBlock(blockIndex, run);

 844    // root is a structure, a pointer to a field in the root would be an alias.
