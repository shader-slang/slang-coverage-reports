 442        /// The number of new output parameters introduced for this output
 443        Index newOutputParamCount = 0;
 444    };

 455    // Parameters can be outputs, so they will also collect information
 456    // into `OutputInfo`s, but they also need additional information

 461    /// A representation of a parameter (possibly an output) of a function
 462    struct ParamInfo : OutputInfo
 463    {
 464        /// Represents what to do with an existing argument at a call site.
 465        enum class OldArgMode

 467            Keep,   ///< Keep the argument as-is.
 468            Ignore, ///< Ignore the argument (eliminate it from the call)

 470        };

 472        /// What do do with existing arguments at call sites
 473        OldArgMode oldArgMode = OldArgMode::Keep;
 474    };

 476    // It is possible that specializing a function output may require
 477    // us to add new output parameters to the function, to enable
 478    // the caller to compute the correct output resource.
 479    //
 480    // For example, consider this input:
 481    //
 482    //      Texture2D getRandomTexture()
 483    //      {
 484    //          int index = /* complicated logic */;
 485    //          return gTextures[index];
 486    //      }
 487    //      ...
 488    //      Texture2D t = getRandomTexture();
 489    //
 490    // The desired output is:
 491    //
 492    //      void _getRandomTexture(out int i)
 493    //      {
 494    //          int index = /* complicated logic */;
 495    //          i = index;
 496    //      }
 497    //      ...
 498    //      int i;
 499    //      getRandomTexture(i);
 500    //      Texture2D t = gTextures[i];
 501    //
 502    // In this case we have made the computation of `t` be
 503    // valid for targets with limited resource support, but
 504    // we have kept the complicated computation of `index`
 505    // in a subroutine, so that we have not bloated the
 506    // code more than necessary.
 507    //
 508    // In order to track new parameters like `i` above,
 509    // we introduce the `NewOutputParamInfo` type.
 510
 511    /// Represents a new output parameter introduced during speicalization
 512    struct NewOutputParamInfo
 513    {
 514        /// The type of the new parameter's *value* (not the pointer type for an `out` parameter)
 515        IRType* type;
 516    };

 518    // Finally, we can aggregate the types above to represent the
 519    // collected information about a function to be specialized.

 521    /// Information about a function to be specialized
 522    struct FuncInfo
 523    {
 524        ReturnValueInfo result;
 525        List<ParamInfo> oldParams;
 526        List<NewOutputParamInfo> newOutputParams;
 527    };

 529    // We now turn to the code that fills in the `FuncInfo` structure.
 530
 531    SpecializeFuncResult specializeFunc(IRFunc* func, FuncInfo& outFuncInfo)
 1532    {
 533        // To specialize a function, we attempt to specialize
 534        // all the applicable parameters and the function result.
 535        //
 536        // Any failures along the way cause the whole process to fail.
 537
 538        // Note: We are introducing new parameters at the same time as we
 539        // iterate over the parameter list, so we cannot just use the
 540        // `func->getParams()` convenience accessor. Instead, we manually
 541        // iterate over the parameters in a way that avoids invalidation
 542        // if we remove the `param` we are working on.
 543        //
 544        // Note: it might seem odd that we are modifying `func` but will
 545        // still bail out on any errors. You might ask: isn't there a chance
 546        // that we will end up with the function in a partially-modified state?
 547        //
 548        // The important thing to remember is that `func` is  *copy* of the
 549        // original function, so any modifications we make to it do not
 550        // affect the original, so that if we *do* have to bail out we can
 551        // leave any call sites intact as calls to the original. The result
 552        // is that bailing out here may leave the new/copied function in
 553        // a state where it isn't useful, but it also won't have any uses,
 554        // and can be eliminated later.
 555        //
 556        IRParam* nextParam = nullptr;
 1557        for (IRParam* param = func->getFirstParam(); param; param = nextParam)
 558        {
 1559            nextParam = param->getNextParam();
 560
 1561            ParamInfo paramInfo;
 1562            auto result = maybeSpecializeParam(param, paramInfo, outFuncInfo);
 1563            if (failedResult(result))
 1564                return result;
 1565            outFuncInfo.oldParams.add(paramInfo);
 566        }
 567
 1568        auto result = maybeSpecializeResult(func, outFuncInfo.result, outFuncInfo);
 1569        if (failedResult(result))
 1570            return result;
 571
 572        return SpecializeFuncResult::Ok;
 1573    }
 574
 575    // The logic for specializing a function result (the return value) is
 576    // simpler than that for parameters, so we will look at it first.
 577
 578    SpecializeFuncResult maybeSpecializeResult(
 579        IRFunc* func,
 580        ReturnValueInfo& outResultInfo,
 581        FuncInfo& ioFuncInfo)
 1582    {
 583        // If the result type of the function isn't a resource type,
 584        // then we don't need to specialize the result, and we
 585        // can succeed without doing anything.
 586        //
 1587        if (!isResourceType(func->getResultType()))
 1588            return SpecializeFuncResult::Ok;
 589
 590        // Otherwise, we know that we will need to produce specialization
 591        // information in `outResultInfo` or fail in the attempt.
 592        //
 593        // We start with the `prepareOutputValue` subroutine which will
 594        // handle some common logic shared with the parameter case.
 595        //
 1596        prepareOutputValue(outResultInfo, ioFuncInfo);
 597
 598        // Next, we want to identify all the places where the function
 599        // `return`s a value, since those establish all the possible
 600        // values for the function result.
 601        //
 602        // Specialization will only be possible if all of those results
 603        // return the "same" value, or values that are in some way
 604        // similar enough for us to collapse into a single pattern.
 605        //
 606        // Identifying the return sites is as simple as looking at
 607        // the terminator instructions of all blocks in the function.
 608        //
 1609        for (auto block : func->getBlocks())
 610        {
 1611            auto returnInst = as<IRReturn>(block->getTerminator());
 1612            if (!returnInst)
 613                continue;
 614
 1615            auto value = returnInst->getVal();
 616
 1617            IRBuilder builder(module);
 1618            builder.setInsertBefore(returnInst);
 619
 620            // Given the `value` being returned, we need to determine
 621            // if it is usable for specializing call sites to this
 622            // function.
 623            //
 624            // If there is a single `return` site, then we can use
 625            // the value returned there as a representative of the
 626            // value returned.
 627            //
 628            // If there are multiple `return` sites, then any sites
 629            // after the first will check if they are similar enough
 630            // in structure to the first one to allow specialization
 631            // to proceed.
 632            //
 633            // If we either fail to identify a specializable result
 634            // or to match a new `return` value against previous
 635            // ones, then the specialization process will fail.
 636            //
 1637            auto result = specializeOutputValue(value, outResultInfo, ioFuncInfo);
 1638            if (failedResult(result))
 1639                return result;
 640
 641            // We will replace the `return <value>;` operation with
 642            // a simple `return;`, because the new specialized function
 643            // will have no return value.
 644            //
 1645            builder.emitReturn();
 1646            returnInst->removeAndDeallocate();
 647        }
 648
 649        // If we have succeeded in gathering information from all
 650        // the `return` sites, then we can finish up computing
 651        // `outResultInfo` and return successfully.
 652        //
 1653        completeOutputValue(outResultInfo, ioFuncInfo);
 1654        return SpecializeFuncResult::Ok;
 1655    }

 657    void prepareOutputValue(OutputInfo& ioValueInfo, FuncInfo& ioFuncInfo)
 1658    {
 659        // This function is called when we have identified that a particular
 660        // value *does* represent an output, but before we have determined
 661        // what value(s) are used for that output.
 662        //
 663        // As such, we set the output into a mode where its value is undefined,
 664        // since that is the approrpiate default to use in the case where
 665        // the function doesn't actually write anything to an output.
 666        //
 667        ioValueInfo.flavor = OutputInfo::Flavor::Undefined;
 668
 669        // We also know that the output might require zero or more new output
 670        // parameters, and we can set the starting index of those parameters
 671        // based on what (if any) has been generated so far.
 672        //
 1673        ioValueInfo.firstNewOutputParamIndex = ioFuncInfo.newOutputParams.getCount();
 1674    }

 676    void completeOutputValue(OutputInfo& ioValueInfo, FuncInfo& ioFuncInfo)
 1677    {
 678        // This function is called when we are done computing the information
 679        // required to specialize a particular output value.
 680        //
 681        // We can now determine how many new output parameters, if any,
 682        // were introduced for the sake of this output.
 683        //
 1684        ioValueInfo.newOutputParamCount =
 685            ioFuncInfo.newOutputParams.getCount() - ioValueInfo.firstNewOutputParamIndex;
 1686    }

 688    SpecializeFuncResult specializeOutputValue(
 689        IRInst* value,
 690        OutputInfo& ioOutputInfo,
 691        FuncInfo& ioFuncInfo)
 1692    {
 693        // This function is called or each `value` that might be written
 694        // to the output identified by `ioOutputInfo`.
 695
 696        // If this is the first call to for the given output, then
 697        // the `representative` value will not have been set.
 698        //
 1699        IRInst* representative = ioOutputInfo.representative;
 1700        if (!representative)
 701        {
 702            // In that case, we will use the given `value` as the
 703            // representative value of this output.
 704            //
 1705            representative = value;
 1706            ioOutputInfo.flavor = OutputInfo::Flavor::Replace;
 1707            ioOutputInfo.representative = representative;
 708        }
 709
 710        // If this is *not* the first call for the given output,
 711        // then we need to confirm that `value` and `representative`
 712        // are suitably matched so that specialization based on `representative`
 713        // will also suffice for `value`.
 714        //
 715        // At the very least, we expect them to be operations with
 716        // the same opcode.
 717        //
 1718        if (value->getOp() != representative->getOp())
 719            return SpecializeFuncResult::ThisFuncFailed;
 720
 721        // Furthermore, only certain instructions are amenable to
 722        // specialization, because in general we cannot reproduce
 723        // an instruction outside of its containing function and
 724        // have it mean the same thing.
 725        //
 726        // We will specifically enumerate the case that we support,
 727        // and expand them over time.
 728        //
 729        // Each supported instruction opcode might introduce new
 730        // constraints on how `value` and `representative` must match.
 731        //
 1732        switch (value->getOp())
 733        {
 734        default:
 735            // Any opcode we do not specifically enable should cause
 736            // specialization to fail.
 737            //
 738            return SpecializeFuncResult::ThisFuncFailed;
 739
 740        case kIROp_GlobalParam:
 741            // A direct reference to a global shader parameter is
 742            // the easiest case to handle.
 743            //
 744            // We do need to require that all values used for the
 745            // same output refer to the *same* global parameter.
 746            //
 1747            if (value != representative)
 748                return SpecializeFuncResult::ThisFuncFailed;
 749            return SpecializeFuncResult::Ok;
 750
 751            // TODO: There are a number of additional cases that we should
 752            // enable here.
 753            //
 754            // The most obvious new cases to support are:
 755            //
 756            // * Function parameters: if the output value is one of the
 757            //   parameter of the function, then callers can just use the
 758            //   same value they passed for the corresponding argument.
 759            //
 760            // * Array indexing: if the array itself is suitable to specialize,
 761            //   then it should be possible to return the array index via
 762            //   a new `out` parameter, and have the caller do the indexing.
 763        }
 764
 765        // Note: the `FuncInfo` is currently being passed in in aid of the
 766        // array-indexing case, but because that case is not implemented
 767        // the parameter is not being used.
 768        //
 769        SLANG_UNUSED(ioFuncInfo);
 770
 771        // TODO: One of the hardest cases here would be `inout` parameters
 772        // of texture type, where the result value depends on the input value(s):
 773        //
 774        //      void swap(inout Texture2D a, inout Texture2D b)
 775        //      {
 776        //          Texture2D tmp = a;
 777        //          a = b;
 778        //          b = tmp;
 779        //      }
 780        //
 781        // In such a case, the value written to `a` is a `load` from parameter `b`,
 782        // but it would be difficult to *prove* that such a load represents the
 783        // value of the parameter on input to the function, rather than on output.
 784        //
 785        // It might be best if resource type legalization replaced `inout`
 786        // parameters of resource type with distinct `in` and `out` parameters,
 787        // to make the relationships more clear.
 1788    }
 789
 790    // As discussed earlier, the case for `out`/`inout` function parameters
 791    // is more involved than that for the function `return` value, so we
 792    // put it off until we'd discussed the shared subroutines.
 793
 794    SpecializeFuncResult maybeSpecializeParam(
 795        IRParam* param,
 796        ParamInfo& outParamInfo,
 797        FuncInfo& ioFuncInfo)
 1798    {
 799        // We only want to specialize in the case where the parameter
 800        // is an `out` or `inout` (both inherit from `IROutParamTypeBase`),
 801        // and the pointed-to type is a resource.
 802        //
 1803        auto paramType = param->getDataType();
 1804        auto outType = as<IROutParamTypeBase>(paramType);
 1805        if (!outType)
 806            return SpecializeFuncResult::Ok;
 1807        auto valueType = outType->getValueType();
 1808        if (!isResourceType(valueType))
 1809            return SpecializeFuncResult::Ok;
 810
 1811        prepareOutputValue(outParamInfo, ioFuncInfo);
 812
 813        // We are going to remove the parameter and add zero or more
 814        // replacements, and we want any replacements to end up
 815        // at the same point in the function signature.
 816        //
 1817        IRBuilder paramsBuilder(module);
 1818        paramsBuilder.setInsertBefore(param);
 819
 820        // We also need to introduce new instructions into the function
 821        // body, as part of the entry block.
 822        //
 1823        IRBlock* block = as<IRBlock>(param->getParent());
 1824        IRBuilder bodyBuilder(module);
 1825        bodyBuilder.setInsertBefore(block->getFirstOrdinaryInst());
 826
 827        // No matter what, we create a local variable that will be
 828        // used to replace the parameter.
 829        //
 1830        IRVar* newVar = bodyBuilder.emitVar(valueType);
 831
 1832        if (as<IRBorrowInOutParamType>(outType))
 833        {
 834            // If the parameter is an `inout` rather than just
 835            // an `out`, then we still need a parameter to
 836            // be passed in, but it can be an `in` parameter
 837            // instead, which means a `T` instead of an
 838            // `InOut<T>`.
 839            //
 1840            IRInst* newParam = paramsBuilder.createParam(valueType);
 1841            newParam->insertBefore(param);
 1842            param->transferDecorationsTo(newParam);
 843
 844            // The start of the function body should assign
 845            // from the `in` parameter to the local variable.
 846            //
 1847            bodyBuilder.emitStore(newVar, newParam);
 848
 849            // We also need call sites to pass in an argument
 850            // for the new `in` parameter, which will have to
 851            // be dereferenced by one level from the original
 852            // argument they were passing.
 853            //
 1854            outParamInfo.oldArgMode = ParamInfo::OldArgMode::Deref;
 1855        }
 856        else
 857        {
 858            // The case for a pure `out` parameter is easier:
 859            // we don't need to initialize the local variable,
 860            // and we don't need callers to pass in anything.
 861            //
 1862            outParamInfo.oldArgMode = ParamInfo::OldArgMode::Ignore;
 863        }
 864
