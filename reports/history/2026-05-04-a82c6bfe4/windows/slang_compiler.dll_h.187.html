 5// This file defines an API for serialization.
 6//
 7// The API is intended to support multiple serialization formats,
 8// and to work with complicated object graphs that may include
 9// shared pointers, circular references, and so on.
 10//
 11// For anybody who don't want to dig into the details, the short
 12// version is that if you have a user-defined type like:
 13//
 14//      // my-thing.h
 15//      ...
 16//
 17//      struct MyThing
 18//      {
 19//          float a;
 20//          List<OtherThing> otherThings;
 21//          SomeObject* object;
 22//      };
 23//
 24// then you can implement serialization support for your type
 25// with something like:
 26//
 27//      template<typename S>
 28//      void serialize(S const& serializer, MyThing& value)
 29//      {
 30//          SLANG_SCOPED_SERIALIZER_STRUCT(serializer);
 31//          serialize(serializer, value.a);
 32//          serialize(serializer, value.otherThings);
 33//          serialize(serializer, value.object);
 34//      }
 35//
 36// That's it. So long as the `OtherThing` and `SomeObject` types used
 37// in the declaration of `MyType` already implemented serialization
 38// support, your new type should be fully serializable.

 458    ///
 459    /// * When writing, if the same pointer value has been seen before,
 460    ///   the `callback` will not be invoked, and instead an additional
 461    ///   reference to the previously-serialized value will be written.
 462    ///
 463    /// * When reading, if the serialized value has been read before,
 464    ///   the `callback` will not be invoked, and instead `value` will
 465    ///   be set to the pointer that was previously read.
 466    ///
 467    virtual void handleSharedPtr(void*& value, Callback callback, void* context) = 0;
 468
 469    /// Defer serialization of the contents of an object.
 470    ///
 471    /// Used to delay serialization of members of an object that
 472    /// could cause infinite recursion if serialized eagerly.
 473    ///
 474    /// This operation should only be used in the body of a callback
 475    /// passed to `handleUniquePtr()` or `handleSharedPtr()`.
 476    ///
 477    /// This operation schedules the given `callback` to be called
 478    /// at some later point a `callback(value, this, context)`, with
 479    /// the state of the serializer implementation restored to what
 480    /// it was when `handleDeferredObjectContents()` was called.
 481    ///
 482    /// Some concrete serializer implementations might implement
 483    /// this operation by invoking `callback` immediately.
 484    ///
 485    virtual void handleDeferredObjectContents(void* value, Callback callback, void* context) = 0;
 486};

 488//
 489// Rather than interface with serialization back-ends directly,
 490// most client code will use a wrapper type that amounts to a kind
 491// of smart pointer.
 492//
 493// While serialization back-ends providing operations comparable to
 494// `ISerializerImpl` above can cover a wide range of
 495// types that need to be serialized, it is common for types to require
 496// more specific *context* to be available in order to perform serialization.
 497// For example, code might need access to a factory object in order
 498// to construct objects of a type being read.
 499//
 500// To support more specialized serializer implementations, the smart
 501// pointer type used for a serializer actually wraps *two* pointers:
 502// one for an `ISerializerImpl`-derived type, and one for a context
 503// type. The smart pointer is templated on both of these types.
 504//
 505
 506/// Base type for serialization contexts.
 507///
 508/// The type parameter `I` should be the serialization back-end
 509/// implementation (e.g., `ISerializerImpl`), while the `C` type
 510/// parameter can be any type that carries additional context
 511/// information needed.
 512///
 513template<typename I, typename C>
 514struct Serializer
 515{
 516public:
 517    using Impl = I;
 518    using Context = C;
 519
 520    SLANG_FORCE_INLINE Serializer() = default;
 521    SLANG_FORCE_INLINE Serializer(Impl* impl, Context* context = nullptr)
 1522        : _impl(impl), _context(context)
 1523    {
 1524    }
 525
 526    template<typename SourceImpl, typename SourceContext>
 527    SLANG_FORCE_INLINE Serializer(
 528        Serializer<SourceImpl, SourceContext> const& serializer,
 529        std::enable_if_t<
 530            std::is_convertible_v<SourceImpl*, Impl*> &&
 531                std::is_convertible_v<SourceContext*, Context*>,
 532            void>* = nullptr)
 533        : _impl(serializer.getImpl()), _context(serializer.getContext())
 534    {
 535    }
 536
 1537    SLANG_FORCE_INLINE Impl* getImpl() const { return _impl; }
 1538    SLANG_FORCE_INLINE Context* getContext() const { return _context; }
 539
 1540    SLANG_FORCE_INLINE Impl* get() const { return _impl; }
 1541    SLANG_FORCE_INLINE Impl* operator->() const { return get(); }
 542
 543
 544private:
 545    Impl* _impl = nullptr;

 979//
 980// Serialization of pointers is the most complicated part of
 981// the whole system. Dealing with pointers means contending with:
 982//
 983// * Multiply-referenced objects, or even cycles in the object graph.
 984//
 985// * Polymoprhic types, where a `Derived*` might get serialized
 986//   through a `Base*` pointer.
 987//
 988// * Types that require going through a factory function of
 989//   some kind as part of their creation (perhaps to implement
 990//   deduplication/caching).
 991//
 992// Our handling of pointers is thus broken down into several
 993// different steps/layers:
 994//
 995// * An ordinary overload of `serialize(s,v)` is used to intercept
 996//   pointer types `T*` and dispatched out to `serializePtr(s,v,(T*)nullptr)`.
 997//   Passing the additional `T*` argument allows different overloads
 998//   of `serializePtr` to intercept entire type hierarchies, while
 999//   still allowing for a fallback case.
 1000//
 1001// * Implementations of `serializePtr` are typically expected to
 1002//   invoke either `serializeUniquePtr` or `serializeSharedPtr`, which
 1003//   handle calling into the `ISerializerImpl` methods with appropriate
 1004//   callbacks.
 1005//
 1006// * The `handleUniquePtr()` or `handleSharedPtr()` operation on
 1007//   `ISerializerImpl` is expected to handle null pointers, or previously-
