 806// Legalizing a uniform buffer/block type for existentials is
 807// more interesting, because we don't actually want to push
 808// the "special" fields out of the buffer entirely (as we
 809// do for resources), and instead we just want to place
 810// them in the buffer *after* all the ordinary data.
 811//
 812// In order to accomplish this we need a way to emit a
 813// constant buffer with a new element type, and then
 814// "wrap" that constant buffer so that it looks like
 815// something that matches the legalization of the original
 816// element type.
 817//
 818// As a concrete example, suppose we have:
 819//
 820//      struct Params { ExistentialBox<Foo> f; int x; ExistentialBox<Bar> b; };
 821//      ConstantBuffer<Params> gParams;
 822//
 823// The legalized form of `Params` will be something like:
 824//
 825//      Pair(
 826//          /* ordinary: */ struct Params_Ordinary { int x; },
 827//          /* special: */ Tuple(
 828//              f -> ImplicitDeref(Foo),
 829//              b -> ImplicitDeref(Bar)))
 830//
 831// We need to be able to splat that all out into a single
 832// structure declaration like:
 833//
 834//      struct Params_Reordered
 835//      {
 836//          Params_Ordinary ordinary;
 837//          Foo f;
 838//          Bar b;
 839//      }
 840//
 841// That allows us to declare:
 842//
 843//      ConstantBuffer<Params_Reordered> gParams;
 844//
 845// That gets the in-memory layout of things correct for the
 846// way we are defining existential value slots to work.
 847// The challenge is that elsewehere in the code there are
 848// operations like `gParams.x` need to now refer to
 849// `gParams.ordinary.x`. Furthermore, even for something like
 850// `f` that seems fine in the example above, we have lost
 851// a level of indirection, so that where we had `load(gParams.f)`
 852// we now want just `gParams.f`.
 853//
 854// The solution is to take `gParams` as soon as it is declared
 855// and wrap it up as a new value:
 856//
 857//      pair(
 858//          /* ordinary: */ gParams.ordinary,
 859//          /* special: */ tuple(
 860//              f -> implicitDeref(gParams.f),
 861//              b -> implicitDeref(gParams.b)))
 862//
 863//
 864// Let's begin by just defining a function that can take
 865// a `LegalType` and turn it into zero or more field
