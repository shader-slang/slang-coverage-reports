 787static void _lookUpInScopes(
 788    ASTBuilder* astBuilder,
 789    Name* name,
 790    LookupRequest const& request,
 791    LookupResult& result)
 1792{
 793    auto thisParameterMode = LookupResultItem::Breadcrumb::ThisParameterMode::Default;
 794
 795    auto scope = request.scope;
 796
 1797    auto endScope = request.endScope;
 798
 799    // The file decl that this scope is in.
 1800    FileDecl* thisFileDecl = nullptr;
 801
 1802    for (; scope != endScope; scope = scope->parent)
 803    {
 804        // Note that we consider all "peer" scopes together,
 805        // so that a hit in one of them does not preclude
 806        // also finding a hit in another
 1807        for (auto link = scope; link; link = link->nextSibling)
 808        {
 1809            auto containerDecl = link->containerDecl;
 810
 811            // It is possible for the first scope in a list of
 812            // siblings to be a "dummy" scope that only exists
 813            // to combine the siblings; in that case it will
 814            // have a null `containerDecl` and needs to be
 815            // skipped over.
 816            //
 1817            if (!containerDecl)
 818                continue;
 819
 1820            if (auto fileDecl = as<FileDecl>(containerDecl))
 821            {
 1822                if (!thisFileDecl)
 1823                    thisFileDecl = fileDecl;
 1824                else if (fileDecl == thisFileDecl)
 825                {
 826                    // If we have already looked up in this file decl,
 827                    // we don't want to do so again.
 828                    continue;
 829                }
 830            }
 831
 832            // TODO: If we need default substitutions to be applied to
 833            // the `containerDecl`, then it might make sense to have
 834            // each `link` in the scope store a decl-ref instead of
 835            // just a decl.
 836            //
 837            DeclRef<ContainerDecl> containerDeclRef = createDefaultSubstitutionsIfNeeded(
 838                                                          astBuilder,
 839                                                          request.semantics,
 840                                                          makeDeclRef(containerDecl))
 1841                                                          .as<ContainerDecl>();
 842
 843            // If the container we are looking into represents a type
 844            // or an `extension` of a type, then we need to treat
 845            // this step as lookup into the `this` variable (or the
 846            // `This` type), which means including any `extension`s
 847            // or inheritance clauses in the lookup process.
 848            //
 849            // Note: The `AggTypeDeclBase` class is the common superclass
 850            // between `AggTypeDecl` and `ExtensionDecl`.
 851            //
 1852            if (auto aggTypeDeclBaseRef = containerDeclRef.as<AggTypeDeclBase>())
 853            {
 854                // When reconstructing the final expression for a result
 855                // looked up through the type or extension, we will need
 856                // a `this` expression (or a `This` type expression) to
 857                // mark the base of the member reference, so we create
 858                // a "breadcrumb" here to track that fact.
 859                //
 1860                BreadcrumbInfo breadcrumb;
 861                breadcrumb.kind = LookupResultItem::Breadcrumb::Kind::This;
 862                breadcrumb.thisParameterMode = thisParameterMode;
 1863                breadcrumb.declRef = aggTypeDeclBaseRef;
 864                breadcrumb.prev = nullptr;
 865                BreadcrumbInfo* breadcrumbPtr = &breadcrumb;
 866                Type* type = nullptr;
 1867                if (auto extDeclRef = aggTypeDeclBaseRef.as<ExtensionDecl>())
 868                {
 1869                    if (request.semantics)
 870                    {
 1871                        ensureDecl(
 872                            request.semantics,
 873                            extDeclRef.getDecl(),
 874                            DeclCheckState::CanUseExtensionTargetType);
 875                    }
 876
 877                    // If we are doing lookup from inside an `extension`
 878                    // declaration, then the `this` expression will have
 879                    // a type that uses the "target type" of the `extension`.
 880                    //
 1881                    type = getTargetType(astBuilder, extDeclRef);
 1882                    if (name == astBuilder->getSharedASTBuilder()->getThisTypeName())
 883                    {
 884                        breadcrumbPtr = nullptr;
 885                    }
 1886                }
 887                else
 888                {
 889                    SLANG_ASSERT(aggTypeDeclBaseRef.as<AggTypeDecl>());
 1890                    if (auto interfaceBase = as<InterfaceDecl>(aggTypeDeclBaseRef.getDecl()))
 891                    {
 892                        // When looking up inside an interface type, we are actually looking up
 893                        // through ThisType.
 1894                        if (name != interfaceBase->getThisTypeDecl()->getName())
 895                        {
 1896                            type = DeclRefType::create(
 897                                astBuilder,
 898                                astBuilder->getMemberDeclRef(
 899                                    aggTypeDeclBaseRef,
 900                                    interfaceBase->getThisTypeDecl()));
 901                            // Don't need any breadcrumb for looking up through ThisType, since we
 902                            // have already created the base type reference in the new `type`'s
 903                            // declref.
 1904                            breadcrumbPtr = nullptr;
 905                        }
 906                    }
 907
 1908                    if (!type)
 909                    {
 1910                        type = DeclRefType::create(astBuilder, aggTypeDeclBaseRef);
 911                    }
 912                }
 913
 914                // When looking up in an extension declaration, we should not automatically
 915                // dereference pointer types, as the 'This' type should refer to the
 916                // extension target type itself, not the pointed-to type.
 1917                LookupRequest modifiedRequest = request;
 1918                if (aggTypeDeclBaseRef.as<ExtensionDecl>())
 919                {
 1920                    modifiedRequest.options = (LookupOptions)((uint32_t)modifiedRequest.options |
 921                                                              (uint32_t)LookupOptions::NoDeref);
 922                }
 923
 1924                _lookUpMembersInType(
 925                    astBuilder,
 926                    name,
 927                    type,
 928                    modifiedRequest,
 929                    result,
 930                    breadcrumbPtr);
 1931            }
 932            else
 933            {
 934                // The default case is when the scope doesn't represent a
 935                // type or `extension` declaration, so we can look up members
 936                // in that scope much more simply.
 937                //
 1938                _lookUpDirectAndTransparentMembers(
 939                    astBuilder,
 940                    name,
 941                    containerDeclRef.getDecl(),
 942                    containerDeclRef,
 943                    request,
 944                    result,
 945                    nullptr);
 946            }
 947
 1948            if (auto defaultImplDecl = as<InterfaceDefaultImplDecl>(containerDecl))
 949            {
 950                // If we are checking an interface default method implementation,
 951                // we should look up members from implicit `this` whose type is the explicit `This`
 952                // generic parameter, and skip looking up in the interface decl itself.
 953
 954                // Instead of looking up in the interface decl itself, we should
 955                // look up in the `This` type instead.
 0956                if (getText(name) != "This")
 957                {
 0958                    BreadcrumbInfo breadcrumb;
 959                    breadcrumb.kind = LookupResultItem::Breadcrumb::Kind::This;
 960                    breadcrumb.thisParameterMode = thisParameterMode;
 0961                    breadcrumb.declRef = DeclRef<Decl>(defaultImplDecl->thisTypeDecl);
 962                    breadcrumb.prev = nullptr;
 0963                    Type* type = DeclRefType::create(astBuilder, breadcrumb.declRef);
 0964                    _lookUpMembersInType(astBuilder, name, type, request, result, &breadcrumb);
 965                }
 966
 967                // We need to skip looking up in the interface decl itself, since we are
 968                // looking up in the implicit `this` type.
 0969                for (; scope && !as<InterfaceDecl>(scope->containerDecl); scope = scope->parent)
 970                {
 971                    // We need to skip looking up in the interface decl itself, since we are
 972                    // looking up in the implicit `this` type.

 975            }
 976
 977            // Before we proceed up to the next outer scope to perform lookup
 978            // again, we need to consider what the current scope tells us
 979            // about how to interpret uses of implicit `this` or `This`. For
 980            // example, if we are inside a `[mutating]` method, then the implicit
