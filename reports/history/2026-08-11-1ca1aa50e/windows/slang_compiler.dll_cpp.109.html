 768template<typename T>
 769static T makeFromSizeVersioned(const uint8_t* src)
 1770{
 771    // The structure size must be size_t
 772    SLANG_COMPILE_TIME_ASSERT(sizeof(((T*)src)->structureSize) == sizeof(size_t));
 773
 774    // The structureSize field *must* be the first element of T
 775    // Ideally would use SLANG_COMPILE_TIME_ASSERT, but that doesn't work on gcc.
 776    // Can't just assert, because determined to be a constant expression
 777    {
 778        auto offset = SLANG_OFFSET_OF(T, structureSize);
 779        SLANG_ASSERT(offset == 0);
 780        // Needed because offset is only 'used' by an assert
 781        SLANG_UNUSED(offset);
 782    }
 783
 784    // The source size is held in the first element of T, and will be in the first bytes of src.
 1785    const size_t srcSize = _getStructureSize(src);
 1786    const size_t dstSize = sizeof(T);
 787
 788    // If they are the same size, and appropriate alignment we can just cast and return
 1789    if (srcSize == dstSize && (size_t(src) & (alignof(T) - 1)) == 0)
 790    {
 1791        return *(const T*)src;
 792    }
 793
 794    // Assumes T can default constructed sensibly
 0795    T dst;
 796
 797    // It's structure size should be setup and should be dstSize
 798    SLANG_ASSERT(dst.structureSize == dstSize);
 799
 800    // The size to copy is the minimum on the two sizes
 0801    const auto copySize = std::min(srcSize, dstSize);
 0802    ::memcpy(&dst, src, copySize);
 803
 804    // The final struct size is the destination size
 805    dst.structureSize = dstSize;
 806
 0807    return dst;
