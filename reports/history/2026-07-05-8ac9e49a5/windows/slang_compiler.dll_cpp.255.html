 758IRInst* tryRemoveTrivialPhi(ConstructSSAContext* context, PhiInfo* phiInfo)
 1759{
 1760    auto phi = phiInfo->phi;
 761
 762    // We are going to check if all of the operands
 763    // to the phi are either the same, or are equal
 764    // to the phi itself.
 765
 1766    IRInst* same = nullptr;
 1767    for (auto u : phiInfo->operands)
 768    {
 1769        auto usedVal = u.get();

 1772        if (usedVal == same || usedVal == phi)
 773        {

 775            // to the same value we've seen already.
 776            continue;
 777        }

 779        {
 780            // We've found at least two distinct values
 781            // other than the phi itself, so this phi
 782            // indeed appears to be non-trivial.

 785            return phi;

 787        else

 789            // This value is distinct from the phi itself,
 790            // so we need to track its value.
 1791            same = usedVal;
 792        }
 793    }
 794
 1795    if (!same)
 796    {
 797        // There were no operands other than the phi itself.
 798        // This implies that the value at the use sites should
 799        // actually be undefined.

 802        // removing the phi node.

 805        // undefined value with the same type as `phi`.

 1813    List<PhiInfo*> otherPhis;

 815    {
 1816        auto user = u->user;

 818            continue;

 820            continue;
 821
 1822        if (user->getOp() == kIROp_Param)
 823        {

 1827                otherPhis.add(otherPhiInfo);

 829        }
 830    }
 831
 832    // replace uses of the phi (including its possible uses
 833    // of itself) with the unique non-phi value.
 1834    phi->replaceUsesWith(same);
 1835    phi->removeAndDeallocate();

 837    // Clear out the operands to the phi, since they won't
 838    // actually get used in the program any more.
 1839    for (auto& u : phiInfo->operands)
 840    {
 1841        u.clear();
 842    }
