 539// If `block` ends in an `ifElse` whose condition is one of `block`'s own parameters
 540// (a phi), and the value that parameter receives along the edge from `fromPred` is a
 541// boolean constant, then only one of the two branches is feasible when arriving from
 542// `fromPred`. Returns the block that is *not* taken (the infeasible successor), or
 543// null if both successors remain feasible.
 544//
 545// This captures the short-circuit `&&`/`||` lowering: the merge block of `a && b`
 546// carries a phi that is the literal `false` along the "a is false" edge, so a later
 547// branch on that phi cannot take its true-side from that edge. Without this, the
 548// flat CFG admits an infeasible store-free path through the merge, producing a false
 549// "possibly uninitialized" warning for patterns like `f(out x) && use(x)`.

 1551{

 590// Remove all loads that are "definitely assigned": every control-flow path from
 591// the function entry to the load passes through at least one store.
 592//
 593// A load is kept (a must-init violation) only when there exists a path from entry
 594// to the load that does not pass through any store first — i.e. the variable can
 595// be read while still uninitialized on at least one path.
 596//
 597// This is the standard definite-assignment property. We compute it directly with a
 598// forward CFG walk from entry that is blocked by store-containing blocks, rather
 599// than using simple dominance. Dominance is too strict: "the load is dominated by
 600// some single store" misses the common-and-safe case where different paths are
 601// guarded by different stores, producing false positives on patterns like
 602// short-circuit `&&` chains (`f(out x) && use(x)`), where the only way to reach the
 603// use is through the store, but no individual store-block dominates the use-block
 604// because the join block has a store-free predecessor whose path never reaches the
 605// use.

 1610{
 1611    if (loads.getCount() == 0)

 619    // the variable somewhere within it, matching the block-reachability granularity the

 1623    HashSet<IRBlock*> wholeBlockStores;

 1625    for (auto store : stores)
 626    {

 633        {
 1634            blocksWithStore.add(block);
 1635            storeSet.add(store);
 636        }

 639    // For blocks that contain both a store and a load, the relative order matters:
 640    // a load that appears before any store in the same block is reachable while
 641    // uninitialized (along the store-free entry path into the block), whereas a load
 642    // after a store in the same block is definitely assigned by that store.

 1646    for (auto load : loads)
 647    {

 652        // is considered definitely assigned.

 672    // Loop relaxation: element-wise initialization inside a loop is extremely common
 673    // (e.g. `[ForceUnroll] for (i) result[i] = ...;` filling an array/vector before
 674    // use). Such a store does not strictly dominate the post-loop use — the loop could

 683    // control reaches the loop-exit (break) block: that break block is never marked

 685    // that reaches it first goes through the loop; excluding it does not hide store-free
 686    // paths that bypass the loop entirely. Clean state still flows into the loop body, so
 687    // uses that precede the store inside the body are still reported.

 698        // initializing the variable by the time control reaches the break block.
 699        //

 701        // fill a variable element-by-element (e.g. `[ForceUnroll] for (i) result[i] =

 704        // post-loop use under a purely structural analysis (the loop "might" run zero
 705        // times, or an inner loop might). Distinguishing those from a genuine
