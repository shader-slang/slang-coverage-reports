 275// A page contains blocks of one specific size (`block_size`).
 276// Each page has three list of free blocks:
 277// `free` for blocks that can be allocated,
 278// `local_free` for freed blocks that are not yet available to `mi_malloc`
 279// `thread_free` for freed blocks by other threads
 280// The `local_free` and `thread_free` lists are migrated to the `free` list
 281// when it is exhausted. The separate `local_free` list is necessary to
 282// implement a monotonic heartbeat. The `thread_free` list is needed for
 283// avoiding atomic operations in the common case.
 284//
 285// `used - |thread_free|` == actual blocks that are in use (alive)
 286// `used - |thread_free| + |free| + |local_free| == capacity`
 287//
 288// We don't count `freed` (as |free|) but use `used` to reduce
 289// the number of memory accesses in the `mi_page_all_free` function(s).
 290//
 291// Notes:
 292// - Access is optimized for `free.c:mi_free` and `alloc.c:mi_page_alloc`
 293// - Using `uint16_t` does not seem to slow things down
 294// - The size is 12 words on 64-bit which helps the page index calculations
 295//   (and 14 words on 32-bit, and encoded free lists add 2 words)
 296// - `xthread_free` uses the bottom bits as a delayed-free flags to optimize
 297//   concurrent frees where only the first concurrent free adds to the owning
 298//   heap `thread_delayed_free` list (see `free.c:mi_free_block_mt`).
 299//   The invariant is that no-delayed-free is only set if there is
 300//   at least one block that will be added, or as already been added, to
 301//   the owning heap `thread_delayed_free` list. This guarantees that pages
 302//   will be freed correctly even if only other threads free blocks.

 307  uint8_t               is_committed:1;    // `true` if the page virtual memory is committed
 308  uint8_t               is_zero_init:1;    // `true` if the page was initially zero initialized
 309  uint8_t               is_huge:1;         // `true` if the page is in a huge segment (`segment->kind == MI_SEGMENT_HUGE
 310                                           // padding
 311  // layout like this to optimize access in `mi_malloc` and `mi_free`
 312  uint16_t              capacity;          // number of blocks committed, must be the first field, see `segment.c:page_c
 313  uint16_t              reserved;          // number of blocks reserved in memory
 314  mi_page_flags_t       flags;             // `in_full` and `has_aligned` flags (8 bits)
 315  uint8_t               free_is_zero:1;    // `true` if the blocks in the free list are zero initialized
 316  uint8_t               retire_expire:7;   // expiration count for retired blocks
