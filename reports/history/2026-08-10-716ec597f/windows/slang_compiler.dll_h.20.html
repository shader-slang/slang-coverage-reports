 146static inline void* mi_prim_tls_slot(size_t slot) mi_attr_noexcept {
 147  void* res;
 148  const size_t ofs = (slot*sizeof(void*));
 149  #if defined(__i386__)
 150    __asm__("movl %%gs:%1, %0" : "=r" (res) : "m" (*((void**)ofs)) : );  // x86 32-bit always uses GS
 151  #elif defined(__APPLE__) && defined(__x86_64__)
 152    __asm__("movq %%gs:%1, %0" : "=r" (res) : "m" (*((void**)ofs)) : );  // x86_64 macOSX uses GS
 153  #elif defined(__x86_64__) && (MI_INTPTR_SIZE==4)
 154    __asm__("movl %%fs:%1, %0" : "=r" (res) : "m" (*((void**)ofs)) : );  // x32 ABI
 155  #elif defined(__x86_64__)
 156    __asm__("movq %%fs:%1, %0" : "=r" (res) : "m" (*((void**)ofs)) : );  // x86_64 Linux, BSD uses FS
 157  #elif defined(__arm__)

 160    res = tcb[slot];
 161  #elif defined(__aarch64__)

 163    #if defined(__APPLE__) // M1, issue #343

 165    #else
 166    __asm__ volatile ("mrs %0, tpidr_el0" : "=r" (tcb));
 167    #endif

 169  #elif defined(__APPLE__) && defined(__POWERPC__) // ppc, issue #781

 280/* ----------------------------------------------------------------------------------------
 281The thread local default heap: `_mi_prim_get_default_heap()`
 282This is inlined here as it is on the fast path for allocation functions.
 283
 284On most platforms (Windows, Linux, FreeBSD, NetBSD, etc), this just returns a
 285__thread local variable (`_mi_heap_default`). With the initial-exec TLS model this ensures
 286that the storage will always be available (allocated on the thread stacks).
 287
 288On some platforms though we cannot use that when overriding `malloc` since the underlying
 289TLS implementation (or the loader) will call itself `malloc` on a first access and recurse.
 290We try to circumvent this in an efficient way:
 291- macOSX : we use an unused TLS slot from the OS allocated slots (MI_TLS_SLOT). On OSX, the
 292           loader itself calls `malloc` even before the modules are initialized.
 293- OpenBSD: we use an unused slot from the pthread block (MI_TLS_PTHREAD_SLOT_OFS).
 294- DragonFly: defaults are working but seem slow compared to freeBSD (see PR #323)
 295------------------------------------------------------------------------------------------- */
 296
 297static inline mi_heap_t* mi_prim_get_default_heap(void);
 298
 299#if defined(MI_MALLOC_OVERRIDE)
 300#if defined(__APPLE__) // macOS
 301  #define MI_TLS_SLOT               89  // seems unused?
 302  // other possible unused ones are 9, 29, __PTK_FRAMEWORK_JAVASCRIPTCORE_KEY4 (94), __PTK_FRAMEWORK_GC_KEY9 (112) and _
 303  // see <https://github.com/rweichler/substrate/blob/master/include/pthread_machdep.h>
 304#elif defined(__OpenBSD__)
 305  // use end bytes of a name; goes wrong if anyone uses names > 23 characters (ptrhread specifies 16)

 307  #define MI_TLS_PTHREAD_SLOT_OFS   (6*sizeof(int) + 4*sizeof(void*) + 24)
 308  // #elif defined(__DragonFly__)

 310  // #define MI_TLS_PTHREAD_SLOT_OFS   (4 + 1*sizeof(void*))  // offset `uniqueid` (also used by gdb?) <https://github.c
