 553// entry-point layouts, so hidden resources such as `ParameterBlock<T>`
 554// members are counted before later target legalization flattens them

 556static bool pickFreeBindingForCoverage(
 557    IRModule* module,

 586    {

 589            return false;

 591        return true;
